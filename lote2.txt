  "/",
  "/product(.*)",
  "/sign-in(.*)",
  "/sign-up(.*)",
  "/termos(.*)",
  "/privacidade(.*)",
  "/cookies(.*)",

  "/api/internal/whatsapp-session(.*)",
]);

const isAdminRoute = createRouteMatcher([
  "/admin(.*)",
]);

export default clerkMiddleware(async (auth, req) => {
  if (!isPublicRoute(req)) {
    await auth.protect();
  }

  if (isAdminRoute(req)) {
    const { userId } = await auth();
    const adminId = process.env.ADMIN_CLERK_USER_ID;

    if (!userId || !adminId || userId !== adminId) {
      return new Response("Acesso não autorizado.", {
        status: 403,
      });
    }
  }
});

import { createHash, createHmac } from "node:crypto";
import prisma from "@/src/lib/prisma";

function gerarDadosTokenAcompanhamento(orderId: string) {
  const secret = process.env.TRACKING_TOKEN_SECRET;

  if (!secret || Buffer.byteLength(secret) < 32) {
    throw new Error("TRACKING_TOKEN_SECRET inválida ou não configurada.");
  }

  const token = createHmac("sha256", secret)
    .update(`agape-order-tracking:${orderId}`)
    .digest("base64url");

  const tokenHash = createHash("sha256")
    .update(token)
    .digest("hex");

  return { token, tokenHash };
}

export async function obterTokenAcompanhamento(orderId: string) {
  const { token, tokenHash } = gerarDadosTokenAcompanhamento(orderId);

  await prisma.order.updateMany({
    where: {
      id: orderId,
      trackingTokenHash: null,
    },
    data: {
      trackingTokenHash: tokenHash,
      trackingTokenCreatedAt: new Date(),
    },
  });

  const pedido = await prisma.order.findUnique({
    where: { id: orderId },
    select: { trackingTokenHash: true },
  });

  if (!pedido || pedido.trackingTokenHash !== tokenHash) {
    throw new Error("Não foi possível gerar o link de acompanhamento.");
  }

  return token;
}

        }

        const pagamentoAtualizado = await tx.order.updateMany({

            id: order.id,

          },
        });

          await tx.orderStatusHistory.create({
            data: {
              orderId: order.id,

              toStatus: "failed",
              source: "stripe_webhook",
            },
          });

          return "failed";
        }

        return "status_not_eligible";
      }

      if (event.type === "payment_intent.canceled") {
        const pedidoAntes = await tx.order.findUnique({
          where: { id: order.id },
          select: { status: true },
        });

        if (!pedidoAntes) {
          throw new Error("Pedido não encontrado antes do cancelamento.");
        }

        if (!["pending", "failed"].includes(pedidoAntes.status)) {
          return "status_not_eligible";
        }

        const pagamentoAtualizado = await tx.order.updateMany({
          where: {
            id: order.id,
            status: pedidoAntes.status,
          },
          data: {
            status: "canceled",
            paymentRecoveryTokenHash: null,
            paymentRecoveryExpiresAt: null,
          },
        });

        if (pagamentoAtualizado.count > 0) {
          await tx.orderStatusHistory.create({
            data: {
              orderId: order.id,
              fromStatus: pedidoAntes.status,
              toStatus: "canceled",
              source: "stripe_webhook",
            },
          });

          return "canceled";
        }

        return "status_not_eligible";
      }

      return "ignored";
    });

    if (pedidoParaNotificar) {
    await processarNotificacaoEmail(
      pedidoParaNotificar,
      "payment_approved"
    );
  }

    if (resultado === "amount_mismatch") {
      return NextResponse.json({
        received: true,
        warning: resultado,
      });
    }

    return NextResponse.json({
      received: true,
      result: resultado,
    });
  } catch (error) {
  if (
    error instanceof Prisma.PrismaClientKnownRequestError &&

        : typeof target === "string" &&
          target.includes("stripeEventId");

    if (duplicidadeDeEvento) {
      return NextResponse.json({
        received: true,
        duplicate: true,
      });
    }
  }

    console.error("Erro ao processar webhook:", error);

    return NextResponse.json(
      { error: "Erro ao processar evento." },
      { status: 500 }
    );
  }
}

import prisma from "@/src/lib/prisma";
import { Prisma } from "@prisma/client";
import { NextResponse } from "next/server";
import { Webhook } from "svix";
import Stripe from "stripe";

type ClerkUserData = {
  id: string;
  first_name?: string | null;
  last_name?: string | null;
  email_addresses?: {
    id: string;
    email_address: string;
  }[];
  primary_email_address_id?: string | null;
  [key: string]: unknown;
};

type ClerkWebhookEvent = {
  type: string;
  data: ClerkUserData;
};

export async function POST(request: Request) {
  const webhookSecret = process.env.CLERK_WEBHOOK_SECRET;

  if (!webhookSecret) {
    console.error("CLERK_WEBHOOK_SECRET não configurado.");
    return NextResponse.json(
      { error: "Configuração do webhook indisponível." },
      { status: 500 }
    );
  }

  // Lê o corpo original, necessário para validar a assinatura.
  const rawBody = await request.text();

  const svixHeaders = {
    "svix-id": request.headers.get("svix-id") ?? "",
    "svix-timestamp": request.headers.get("svix-timestamp") ?? "",
    "svix-signature": request.headers.get("svix-signature") ?? "",
  };

  if (
    !svixHeaders["svix-id"] ||
    !svixHeaders["svix-timestamp"] ||
    !svixHeaders["svix-signature"]
  ) {
    return NextResponse.json(
      { error: "Cabeçalhos de assinatura ausentes." },
      { status: 400 }
    );
  }

  let event: ClerkWebhookEvent;

  try {
    const webhook = new Webhook(webhookSecret);

webhook.verify(rawBody, svixHeaders);

// A assinatura foi validada. Agora interpretamos o JSON original.
event = JSON.parse(rawBody) as ClerkWebhookEvent;
  } catch (error) {
    console.error("Falha na validação do webhook Clerk:", error);

    return NextResponse.json(
      { error: "Assinatura inválida." },
      { status: 400 }
    );
  }

  // Só precisamos processar eventos de criação e atualização.

  // Processa a exclusão solicitada pelo cliente.
  if (event.type === "user.deleted") {
    const deletedUserId = event.data?.id;
    const eventId = svixHeaders["svix-id"];

    if (!deletedUserId) {
      return NextResponse.json(
        { error: "Identificador do usuário ausente." },
        { status: 400 }
      );
    }

    try {
      await prisma.$transaction(async (tx) => {
        const existingUser = await tx.user.findUnique({
          where: { externalId: deletedUserId },
        });

        const deletionRequestedAt =
          existingUser?.deletionRequestedAt ?? new Date();

        const user = await tx.user.upsert({
          where: { externalId: deletedUserId },
          create: {
            externalId: deletedUserId,
            attributes: {},
            deletionRequestedAt,
          },
          update: {
            attributes: {},
            deletionRequestedAt,
          },
        });

        await tx.privacyRequest.createMany({
          data: [
            {
              userId: user.id,
              eventId,
              type: "EXCLUSAO_CONTA",
              source: "CLERK_WEBHOOK",
            },
          ],
          skipDuplicates: true,
        });
      });

      return NextResponse.json(
        { received: true, deletionRequested: true },
        { status: 200 }
      );
    } catch (error) {
      console.error("Erro ao registrar exclusão da conta:", error);

      return NextResponse.json(
        { error: "Falha ao registrar exclusão." },
        { status: 500 }
      );
    }
  }


    // Os demais eventos não precisam ser processados.
    if (
    event.type !== "user.created" &&
    event.type !== "user.updated"
    ) {
    return NextResponse.json({ received: true }, { status: 200 });
    }

  const {
    id,
    first_name,
    last_name,
    email_addresses,
    primary_email_address_id,
    ...attributes
  } = event.data;

  if (!id) {
    return NextResponse.json(
      { error: "Identificador do usuário ausente." },
      { status: 400 }
    );
  }

  const stripeSecretKey = process.env.STRIPE_SECRET_KEY;

  if (!stripeSecretKey) {
    console.error("STRIPE_SECRET_KEY não configurado.");

    return NextResponse.json(
      { error: "Configuração de pagamento indisponível." },
      { status: 500 }
    );
  }

  const stripe = new Stripe(stripeSecretKey, {
    apiVersion: "2026-09-30.endive",
  });

  try {
    const email =
      email_addresses?.find(
        (item) => item.id === primary_email_address_id
      )?.email_address ??
      email_addresses?.[0]?.email_address;

    const name = [first_name, last_name]
      .filter(Boolean)
      .join(" ");

    const existingUser = await prisma.user.findUnique({
    where: { externalId: id },
    });

    if (existingUser?.deletionRequestedAt) {
    return NextResponse.json(
        { received: true, ignored: "account_deletion_requested" },
        { status: 200 }
    );
    }

    let stripeCustomerId = existingUser?.stripeCustomerId;

    // Cria um cliente Stripe apenas se ainda não existir um vínculo.
    if (!stripeCustomerId) {
      const customer = await stripe.customers.create(
        {
          metadata: {
            clerkUserId: id,
          },
        },
        {
          idempotencyKey: `clerk-user-${id}`,
        }
      );

      stripeCustomerId = customer.id;
    }

    // Atualiza os dados do cliente sem criar outro registro.
    const customerUpdate: Stripe.CustomerUpdateParams = {};

    if (name) {
      customerUpdate.name = name;
    }

    if (email) {
      customerUpdate.email = email;
    }

    if (Object.keys(customerUpdate).length > 0) {
      await stripe.customers.update(
        stripeCustomerId,
        customerUpdate
      );
    }

    await prisma.user.upsert({
      where: {
        externalId: id,
      },
      create: {
        externalId: id,
        stripeCustomerId,
        attributes: attributes as Prisma.InputJsonObject,
      },
      update: {
        stripeCustomerId,
        attributes: attributes as Prisma.InputJsonObject,
      },
    });

    return NextResponse.json(
      { received: true },
      { status: 200 }
    );
  } catch (error) {
    console.error("Erro ao processar usuário do Clerk:", error);

    // Retorna 500 para permitir que o Clerk tente entregar novamente.
    return NextResponse.json(
      { error: "Falha ao processar usuário." },
      { status: 500 }
    );
  }
}

import { createHash } from "node:crypto";
import { connection, NextResponse } from "next/server";
import prisma from "@/src/lib/prisma";

function respostaJson(
  dados: unknown,
  opcoes: ResponseInit = {}
) {
  return NextResponse.json(dados, {
    ...opcoes,
    headers: {
      "Cache-Control": "no-store, private",
      "Referrer-Policy": "no-referrer",
      "X-Content-Type-Options": "nosniff",
    },
  });
}

function gerarHashToken(token: string) {
  return createHash("sha256").update(token).digest("hex");
}

export async function GET(request: Request) {
  await connection();
  try {
    const url = new URL(request.url);
    const token = url.searchParams.get("token");

    if (
      !token ||
      token.length !== 43 ||
      !/^[A-Za-z0-9_-]+$/.test(token)
    ) {
      return respostaJson(
        { error: "Link de acompanhamento invÃ¡lido ou expirado." },
        {
          status: 404,
          headers: { "Cache-Control": "no-store" },
        }
      );

import { NextRequest, NextResponse } from "next/server";
import crypto from "node:crypto";
import prisma from "../../../../lib/prisma";

const TIPOS_PERMITIDOS = new Set([
  "pre-key",
  "session",
  "sender-key",
  "sender-key-memory",
  "app-state-sync-key",
  "app-state-sync-version",
  "lid-mapping",
  "device-list",
  "tctoken",
  "identity-key",
]);

function obterConfiguracao() {
  const secret = process.env.WHATSAPP_SERVICE_SECRET;
  const encryptionKey = process.env.WHATSAPP_SESSION_ENCRYPTION_KEY;

  if (!secret || secret.length < 40) {
    throw new Error("WHATSAPP_SERVICE_SECRET não configurado.");
  }

  if (!encryptionKey) {
    throw new Error("Chave de criptografia não configurada.");
  }

  const key = Buffer.from(encryptionKey, "base64");

  if (key.length !== 32) {
    throw new Error("Chave de criptografia inválida.");
  }

  return { secret, key };
}

function autorizado(request: NextRequest, secret: string) {
  const recebido = Buffer.from(
    request.headers.get("authorization") || ""
  );

  const esperado = Buffer.from(`Bearer ${secret}`);

  return (
    recebido.length === esperado.length &&
    crypto.timingSafeEqual(recebido, esperado)
  );
}

function criptografar(valor: unknown, key: Buffer) {
  const iv = crypto.randomBytes(12);

  const cipher = crypto.createCipheriv(
    "aes-256-gcm",
    key,
    iv
  );

  const texto = JSON.stringify(valor);

  const encrypted = Buffer.concat([
    cipher.update(texto, "utf8"),
    cipher.final(),
  ]);

  const tag = cipher.getAuthTag();

  return [
    "v1",
    iv.toString("base64"),
    tag.toString("base64"),
    encrypted.toString("base64"),
  ].join(":");
}

function descriptografar(valor: string, key: Buffer) {
  const partes = valor.split(":");

  if (partes.length !== 4 || partes[0] !== "v1") {
    throw new Error("Formato de sessão inválido.");
  }

  const iv = Buffer.from(partes[1], "base64");
  const tag = Buffer.from(partes[2], "base64");
  const encrypted = Buffer.from(partes[3], "base64");

  if (iv.length !== 12 || tag.length !== 16) {
    throw new Error("Dados criptográficos inválidos.");
  }

  const decipher = crypto.createDecipheriv(
    "aes-256-gcm",
    key,
    iv
  );

  decipher.setAuthTag(tag);

  const texto = Buffer.concat([
    decipher.update(encrypted),
    decipher.final(),
  ]).toString("utf8");

  return JSON.parse(texto);
}

async function verificar(request: NextRequest) {
  const { secret, key } = obterConfiguracao();

  if (!autorizado(request, secret)) {
    return null;
  }

  return key;
}

export async function GET(request: NextRequest) {
  try {
    const key = await verificar(request);

    if (!key) {
      return NextResponse.json(
        { error: "Não autorizado." },
        { status: 401 }
      );
    }

    const params = request.nextUrl.searchParams;
    const resource = params.get("resource");

    if (resource === "creds") {
      const registro = await prisma.whatsAppSession.findUnique({
        where: { id: "baileys" },
      });

      return NextResponse.json({
        value: registro
          ? descriptografar(registro.encryptedCreds, key)
          : null,
      });
    }

    if (resource === "keys") {
      const type = params.get("type");
      const ids = params.getAll("id");

      if (
        !type ||
        !TIPOS_PERMITIDOS.has(type) ||
        ids.length > 100 ||
        ids.some((id) => !id || id.length > 500)
      ) {
        return NextResponse.json(
          { error: "Consulta de chaves inválida." },
          { status: 400 }
        );
      }

      if (ids.length === 0) {
        return NextResponse.json({ value: {} });
      }

      const registros = await prisma.whatsAppSessionKey.findMany({
        where: {
          keyType: type,
          keyId: { in: ids },
        },
      });

      const value: Record<string, unknown> = {};

      for (const registro of registros) {
        value[registro.keyId] = descriptografar(
          registro.encryptedData,
          key
        );
      }

      return NextResponse.json({ value });
    }

    return NextResponse.json(
      { error: "Recurso inválido." },
      { status: 400 }
    );
  } catch {
    return NextResponse.json(
      { error: "Não foi possível consultar a sessão." },
      { status: 500 }
    );
  }
}

export async function POST(request: NextRequest) {
  try {
    const key = await verificar(request);

    if (!key) {
      return NextResponse.json(
        { error: "Não autorizado." },
        { status: 401 }
      );
    }

    const texto = await request.text();

    if (Buffer.byteLength(texto, "utf8") > 1_000_000) {
      return NextResponse.json(
        { error: "Dados muito grandes." },
        { status: 413 }
      );
    }

    const body = JSON.parse(texto);

    if (body?.resource === "creds") {
      if (
        !body.value ||
        typeof body.value !== "object" ||
        Array.isArray(body.value)
      ) {
        return NextResponse.json(
          { error: "Credenciais inválidas." },
          { status: 400 }
        );
      }

      await prisma.whatsAppSession.upsert({
        where: { id: "baileys" },
        create: {
          id: "baileys",
          encryptedCreds: criptografar(body.value, key),
        },
        update: {
          encryptedCreds: criptografar(body.value, key),
        },
      });

      return NextResponse.json({ status: "saved" });
    }

    if (body?.resource === "keys") {
      const values = body.values;

      if (
        !values ||
        typeof values !== "object" ||
        Array.isArray(values)
      ) {
        return NextResponse.json(
          { error: "Chaves inválidas." },
          { status: 400 }
        );
      }

      const tipos = Object.keys(values);

      if (
        tipos.length > TIPOS_PERMITIDOS.size ||
        tipos.some((type) => !TIPOS_PERMITIDOS.has(type))
      ) {
        return NextResponse.json(
          { error: "Tipo de chave inválido." },
          { status: 400 }
        );
      }

      await prisma.$transaction(async (tx) => {
        for (const type of tipos) {
          const entries = values[type];

          if (
            !entries ||
            typeof entries !== "object" ||
            Array.isArray(entries) ||
            Object.keys(entries).length > 100
          ) {
            throw new Error("Conjunto de chaves inválido.");
          }

          for (const [id, value] of Object.entries(entries)) {
            if (!id || id.length > 500) {
              throw new Error("Identificador inválido.");
            }

            if (value === null) {
              await tx.whatsAppSessionKey.deleteMany({
                where: { keyType: type, keyId: id },
              });

              continue;
            }

            const encryptedData = criptografar(value, key);

            await tx.whatsAppSessionKey.upsert({
              where: {
                keyType_keyId: {
                  keyType: type,
                  keyId: id,
                },
              },
              create: {
                keyType: type,
                keyId: id,
                encryptedData,
              },
              update: { encryptedData },
            });
          }
        }
      });

      return NextResponse.json({ status: "saved" });
    }

    return NextResponse.json(
      { error: "Recurso inválido." },
      { status: 400 }
    );
  } catch {
    return NextResponse.json(
      { error: "Não foi possível salvar a sessão." },
      { status: 500 }
    );
  }
}

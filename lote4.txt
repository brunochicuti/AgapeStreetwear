async function validarCepEntrega(
  dados: DadosComprador
): Promise<string | null> {
  const cep = dados.zipCode.replace(/\D/g, "");

  try {
    const response = await fetch(
      `https://viacep.com.br/ws/${cep}/json/`,
      {
        signal: AbortSignal.timeout(5000),
        cache: "no-store",
        headers: {
          Accept: "application/json",
        },
      }
    );

    if (!response.ok) {
      return "Não foi possível validar o CEP. Tente novamente.";
    }

    const resultado: unknown = await response.json();

    if (
      !resultado ||
      typeof resultado !== "object" ||
      Array.isArray(resultado)
    ) {
      return "Não foi possível validar o CEP. Tente novamente.";
    }

    const endereco = resultado as Record<string, unknown>;

    if (endereco.erro === true) {
      return "CEP não encontrado.";
    }

    if (
      typeof endereco.cep !== "string" ||
      typeof endereco.localidade !== "string" ||
      typeof endereco.uf !== "string"
    ) {
      return "Não foi possível validar o CEP. Tente novamente.";
    }

    if (endereco.cep.replace(/\D/g, "") !== cep) {
      return "Não foi possível confirmar o CEP informado.";
    }

    const normalizar = (valor: string) =>
      valor
        .trim()
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase();

    const cidadeCep = normalizar(endereco.localidade);
    const cidadeComprador = normalizar(dados.city);
    const ufCep = endereco.uf.trim().toUpperCase();
    const ufComprador = dados.state.trim().toUpperCase();

    if (
      ufCep !== "MS" ||
      !["dourados", "maracaju"].includes(cidadeCep)
    ) {
      return "Entrega disponível apenas para Dourados/MS e Maracaju/MS.";
    }

    if (cidadeComprador !== cidadeCep || ufComprador !== ufCep) {
      return "A cidade ou o estado informado não corresponde ao CEP.";
    }

    return null;
  } catch {
    return "Não foi possível validar o CEP. Tente novamente.";
  }
}

import { createHmac } from "node:crypto";
import { isIP } from "node:net";
import prisma from "@/src/lib/prisma";

const JANELA_MS = 10 * 60 * 1000;
const LIMITE_USUARIO = 5;
const LIMITE_IP = 15;

type ResultadoContagem = {
  count: number;
  windowStartedAt: Date;
};

function gerarChave(tipo: "user" | "ip", valor: string) {
  const secret = process.env.RATE_LIMIT_SECRET;

  if (!secret || Buffer.byteLength(secret) < 32) {
    throw new Error("RATE_LIMIT_SECRET ausente ou inválida.");
  }

  const hash = createHmac("sha256", secret)
    .update(`${tipo}:${valor}`)
    .digest("hex");

  return `${tipo}:${hash}`;
}

function obterIP(request: Request) {
  const ip = request.headers.get("x-real-ip")?.trim();

  if (ip && isIP(ip)) return ip;

  if (process.env.NODE_ENV !== "production") {
    return "127.0.0.1";
  }

  throw new Error("Nao foi possivel identificar o IP da requisicao.");
}

async function incrementarContador(key: string) {
  const rows = await prisma.$queryRaw<ResultadoContagem[]>`
    INSERT INTO "CheckoutRateLimit" ("key", "count", "windowStartedAt")
    VALUES (${key}, 1, NOW())
    ON CONFLICT ("key") DO UPDATE SET
      "count" = CASE
        WHEN "CheckoutRateLimit"."windowStartedAt"
          <= NOW() - INTERVAL '10 minutes'
        THEN 1
        ELSE "CheckoutRateLimit"."count" + 1
      END,
      "windowStartedAt" = CASE
        WHEN "CheckoutRateLimit"."windowStartedAt"
          <= NOW() - INTERVAL '10 minutes'
        THEN NOW()
        ELSE "CheckoutRateLimit"."windowStartedAt"
      END
    RETURNING "count", "windowStartedAt"
  `;

  if (!rows[0]) {
    throw new Error("Não foi possível atualizar o contador.");
  }

  return rows[0];
}

function calcularEspera(inicio: Date) {
  return Math.max(
    1,
    Math.ceil((inicio.getTime() + JANELA_MS - Date.now()) / 1000)
  );
}

export async function verificarRateLimitCheckout(
  request: Request,
  userId: string
) {
  const ip = obterIP(request);

  const chaveUsuario = gerarChave("user", userId);
  const chaveIP = gerarChave("ip", ip);

  const usuario = await incrementarContador(chaveUsuario);
  const endereco = await incrementarContador(chaveIP);

  if (Math.random() < 0.01) {
    try {
      await prisma.$executeRaw`
        DELETE FROM "CheckoutRateLimit"
        WHERE "windowStartedAt" < NOW() - INTERVAL '1 day'
      `;
    } catch {
      // A limpeza é secundária; a contagem já foi registrada.
    }
  }

  const bloqueios = [];

  if (usuario.count > LIMITE_USUARIO) {
    bloqueios.push(calcularEspera(usuario.windowStartedAt));
  }

  if (endereco.count > LIMITE_IP) {
    bloqueios.push(calcularEspera(endereco.windowStartedAt));
  }

  return {
    allowed: bloqueios.length === 0,
    retryAfterSeconds: bloqueios.length
      ? Math.max(...bloqueios)
      : 0,
  };
}

  await connection();

  try {
    if (!(await verificarAdmin())) {
      return Response.json(
        { error: "Acesso não autorizado." },
        { status: 403 }
      );
    }

    const { userId } = await auth();

    if (!userId) {
      return Response.json(
        { error: "Usuário não autenticado." },
        { status: 401 }
      );
    }

    const body = await request.json();

    const id = body.id;
    const novoStatus = body.status;
    const decisao = body.decision;
    const justificativa = body.justification;

    if (
      typeof id !== "string" ||
      !id.trim() ||
      typeof novoStatus !== "string" ||
      !statusPermitidos.includes(
        novoStatus as (typeof statusPermitidos)[number]
      )
    ) {
      return Response.json(
        { error: "Dados inválidos." },
        { status: 400 }
      );
    }

    if (
      statusFinais.includes(novoStatus) &&
      (
        typeof decisao !== "string" ||
        !decisao.trim() ||
        typeof justificativa !== "string" ||
        !justificativa.trim()
      )
    ) {
      return Response.json(
        {
          error:
            "Informe a decisão e a justificativa para concluir a solicitação.",
        },
        { status: 400 }
      );
    }

    const solicitacao = await prisma.privacyRequest.findUnique({
      where: { id },
      select: { id: true, status: true },
    });

    if (!solicitacao) {
      return Response.json(
        { error: "Solicitação não encontrada." },
        { status: 404 }
      );
    }

    const transicoesPermitidas: Record<string, string[]> = {
      PENDENTE: ["EM_ANALISE"],
      EM_ANALISE: ["CONCLUIDA", "PARCIAL", "INDEFERIDA"],
    };

      return Response.json(
        { error: "Transição de status não permitida." },
        { status: 409 }
      );
    }

    const resultado = await prisma.privacyRequest.updateMany({
      where: {
        id,
        status: solicitacao.status,
      },
      data: {
        status: novoStatus,
        decision:
          typeof decisao === "string" ? decisao.trim() : null,
        justification:
          typeof justificativa === "string"
            ? justificativa.trim()
            : null,
        processedBy: userId,
        completedAt: statusFinais.includes(novoStatus)
          ? new Date()
          : null,
      },
    });

    if (resultado.count === 0) {
      return Response.json(
        {
          error:
            "A solicitação foi alterada por outra operação. Atualize a página.",
        },
        { status: 409 }
      );

import { NextResponse } from "next/server";
import prisma from "@/src/lib/prisma";

function erroJson(error: string, status: number) {
  return NextResponse.json({ error }, { status });
}

export async function POST(request: Request) {
  try {
    // Confirma a autenticação do cliente.
    const { userId } = await auth();

    if (!userId) {
      return erroJson("Não autenticado.", 401);
    }

    // Limita o tamanho da requisição.
    const contentLength = request.headers.get("content-length");

    if (contentLength && Number(contentLength) > 2048) {
      return erroJson("Requisição muito grande.", 413);
    }

    let body: unknown;

    try {
      const texto = await request.text();

      if (new TextEncoder().encode(texto).length > 2048) {
        return erroJson("Requisição muito grande.", 413);
      }

      body = JSON.parse(texto);
    } catch {
      return erroJson("Requisição inválida.", 400);
    }

    if (
      !body ||
      typeof body !== "object" ||
      Array.isArray(body)
    ) {
      return erroJson("Dados inválidos.", 400);
    }

    const dados = body as Record<string, unknown>;
    const paymentIntentId = dados.paymentIntentId;
    const paymentMethod = dados.paymentMethod;

    if (
      typeof paymentIntentId !== "string" ||
      !/^pi_[a-zA-Z0-9]+$/.test(paymentIntentId)
    ) {
      return erroJson("Identificação do pagamento inválida.", 400);
    }

    if (!["pix", "card"].includes(String(paymentMethod))) {
      return erroJson("Método de pagamento inválido.", 400);
    }

    // Localiza a conta interna correspondente ao usuário autenticado.
    const user = await prisma.user.findUnique({
      where: {
        externalId: userId,
      },
      select: {
        id: true,
        deletionRequestedAt: true,
      },
    });

    if (!user || user.deletionRequestedAt) {
      return erroJson("Usuário não autorizado.", 403);
    }

    // Consulta o pagamento diretamente na Stripe.
    const paymentIntent =
      await stripe.paymentIntents.retrieve(paymentIntentId);

    if (
      paymentIntent.metadata.userId !== userId ||
      !paymentIntent.metadata.orderId ||
      paymentIntent.status !== "requires_payment_method" ||
      paymentIntent.currency !== "brl"
    ) {
      return erroJson(
        "Não foi possível validar o pagamento.",
        409
      );
    }

    // Confirma que o pagamento pertence ao pedido e ao usuário.
    const order = await prisma.order.findFirst({
      where: {
        id: paymentIntent.metadata.orderId,
        userId: user.id,
        paymentIntentID: paymentIntent.id,
        status: {
          in: ["pending", "failed"],
        },
        paymentReminderSentAt: null,
      },
      select: {
        id: true,
        amount: true,
      },
    });

    if (!order) {
      return erroJson(
        "Pedido não encontrado ou não disponível para atualização.",
        409
      );
    }

    // Confere o valor antes de atualizar o método.
    if (Math.round(order.amount * 100) !== paymentIntent.amount) {
      return erroJson(
        "O valor do pagamento não corresponde ao pedido.",
        409
      );
    }

    const atualizado = await prisma.order.updateMany({
      where: {
        id: order.id,
        userId: user.id,
        paymentIntentID: paymentIntent.id,
        status: {
          in: ["pending", "failed"],
        },
        paymentReminderSentAt: null,
      },
      data: {
        paymentMethod: paymentMethod as string,
      },
    });

      return erroJson(
        "Não foi possível atualizar o método de pagamento.",
        409
      );
    }

    return NextResponse.json({
      success: true,
    });
  } catch (error) {

      "Erro ao registrar método de pagamento:",

  }
    finally {
    if (orderId && lockId) {
      await prisma.order.updateMany({
        where: {
          id: orderId,
          paymentRecoveryLockId: lockId,
        },
        data: {
          paymentRecoveryLockId: null,
          paymentRecoveryLockExpiresAt: null,
        },
      });
    }
  }
}
